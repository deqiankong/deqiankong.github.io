# Setup omitted: A is the adjacency matrix of the free cells of the maze,
# deg its row sums, and P = I/2 + A/(2 deg) the lazy random walk on them.

def q(Pt):
    """Normalized transition probability, so that q(x | x, t) = 1."""
    K = Pt / deg[None, :]        # symmetric kernel of the reversible walk
    d = np.sqrt(np.diag(K))
    return K / np.outer(d, d)

# Scales t = 1, 2, 4, ...: P_2t = P_t squared.
scales, Pt = [], P.copy()
for k in range(12):
    scales.append(q(Pt))
    Pt = Pt @ Pt

def plan(x, y):
    """Greedy ascent on q(y | x, t), at the scale with the largest gain."""
    path = [x]
    while x != y:
        nbrs = np.nonzero(A[x])[0]
        gain = np.array([[Q[n, y] - Q[x, y] for n in nbrs] for Q in scales])
        x = nbrs[gain.max(0).argmax()]
        path.append(x)
    return path
